📊 Sincronización con Google Sheets

Conecta tu hoja de cálculo de Google para guardar automáticamente cada pedido nuevo, actualizar estados y respaldar tu tienda en Railway sin base de datos SQL.

📖 Guía paso a paso para conectar Google Sheets con Fórmulas
  1. Abre una hoja nueva en Google Sheets y ve al menú superior: Extensiones > Apps Script.
  2. Borra cualquier código existente en el editor de Apps Script y pega el contenido del script oficial.
  3. Haz clic en Implementar > Nueva implementación (ícono de engranaje > Aplicación web).
  4. En Quién tiene acceso (Who has access) selecciona obligatoriamente Cualquier persona (Anyone).
  5. Haz clic en Implementar, autoriza los permisos de Google y copia la URL terminada en /exec.
  6. Pega la URL arriba en este panel y haz clic en Guardar y Conectar. ¡Se crearán automáticamente las pestañas Dashboard (con todas las fórmulas), Pedidos y Productos!

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Pedidos') || ss.getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var orders = [];
  for (var i = 1; i < rows.length; i++) {
    var raw = rows[i][14];
    if (raw) { try { orders.push(JSON.parse(raw)); } catch(err) {} }
  }
  return ContentService.createTextOutput(JSON.stringify({ orders: orders }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var body = JSON.parse(e.postData.contents || '{}');
  var action = body.action;

  var ordersSheet = getOrCreateSheet(ss, 'Pedidos', [
    'ID Pedido','Fecha','Estado','Cliente','DNI','Telefono','Distrito','Provincia','Departamento','Direccion','Referencia','Productos','Total','Metodo Envio','JSON_RAW'
  ]);
  var productsSheet = getOrCreateSheet(ss, 'Productos', [
    'ID Producto','Titulo','Marca','Categoria','Precio Venta (S/.)','Precio Regular (S/.)','Precio Bundle (S/.)','Stock','Estado','Imagen URL','Descripcion','JSON_RAW'
  ]);

  if (action === 'create' || action === 'update') {
    upsertOrderRow(ordersSheet, body.order);
  } else if (action === 'delete' && body.id) {
    deleteRowById(ordersSheet, body.id);
  } else if (action === 'sync_all' && Array.isArray(body.orders)) {
    body.orders.forEach(function(o) { upsertOrderRow(ordersSheet, o); });
  } else if (action === 'create_product' || action === 'update_product') {
    upsertProductRow(productsSheet, body.product);
  } else if (action === 'delete_product' && body.id) {
    deleteRowById(productsSheet, body.id);
  } else if (action === 'sync_products' && Array.isArray(body.products)) {
    body.products.forEach(function(p) { upsertProductRow(productsSheet, p); });
  }

  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet(ss, name, headers) {
  var s = ss.getSheetByName(name);
  if (!s) {
    s = ss.insertSheet(name);
    s.appendRow(headers);
  } else if (s.getLastRow() === 0) {
    s.appendRow(headers);
  }
  return s;
}

function upsertOrderRow(sheet, o) {
  if (!o || !o.id) return;
  var items = (o.items || []).map(function(i){ return i.quantity + 'x ' + i.title; }).join(', ');
  var rowData = [
    o.id, o.createdAt || '', o.status || 'Pendiente',
    o.customer?.name || '', o.customer?.dni || '', o.customer?.phone || '',
    o.customer?.district || '', o.customer?.province || '', o.customer?.department || '',
    o.customer?.address || '', o.customer?.reference || '',
    items, o.total || 0, o.shippingRate?.name || '', JSON.stringify(o)
  ];
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === o.id) {
      sheet.getRange(i + 1, 1, 1, rowData.length).setValues([rowData]);
      return;
    }
  }
  sheet.appendRow(rowData);
}

function upsertProductRow(sheet, p) {
  if (!p || !p.id) return;
  var rowData = [
    p.id, p.title || '', p.brand || '', p.category || '',
    p.price || 0, p.oldPrice || 0, p.bundlePrice || 0,
    p.stockLeft || 0, p.isSoldOut ? 'Agotado' : 'Disponible',
    (p.images && p.images[0]) || '', p.description || '', JSON.stringify(p)
  ];
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === p.id) {
      sheet.getRange(i + 1, 1, 1, rowData.length).setValues([rowData]);
      return;
    }
  }
  sheet.appendRow(rowData);
}

function deleteRowById(sheet, id) {
  var data = sheet.getDataRange().getValues();
  for (var i = data.length - 1; i >= 1; i--) {
    if (data[i][0] === id) { sheet.deleteRow(i + 1); return; }
  }
}
